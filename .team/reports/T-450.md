import { Circuit } from './upstream/src/runtime/numeric-ui-core.ts';
const circuit = new Circuit(3);
const x = circuit.source(0);
const a = circuit.source(1);
const b = circuit.source(2);
const y = circuit.affine([[circuit.multiply(a, x), 1], [b, 1]]);
const source = new Float32Array([20, 1.25, -2]);
const values = circuit.evaluateReference(source);
