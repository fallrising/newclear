// 交錯進場（Stagger）
// 分類組: motion (動態圖像與動畫效果) · element 02 · 分類: 登場與文字
// 示範內容: 圖表長條由左至右依序升起，每根略有延遲。
// 動作／視覺規格:
//   - 每 2.35 s 一輪。七根長條（寬 44、間距 20、高度為 52, 84, 68, 112, 92, 136, 158），依序輪用藍、珊瑚、黃、薄荷綠。
//   - 第 i 根長條（從 0 起算）於 0.15 + 0.12·i s 開始，以底線為錨點，在 0.5 s 內從 0 長至完整高度，使用 backOut（s = 2.6），超出後回穩。
//   - 白點（半徑 6.5，位於長條頂端上方 16 px）：比長條晚 0.34 s 開始，彈簧曲線為 1 − e^(−7t)·cos(16t)。
//   - 退場：長條從 1.78 + 0.045·i s 開始，以三次方 ease-in 在 0.26 s 內縮回；白點先從 1.70 + 0.045·i s 開始，在 0.2 s 內消失。
// 使用情境: 適合清單列、圖表長條或卡片等多個元素，依序進場以引導視線。
// 關鍵字: stagger, sequence, delay, cascade, bar-chart, list, entrance
// 尺寸: 539x208 px · 週期: 4.7 s
// 需要: kit motion -> ./api/v1/kit/zh-Hant/motion.js
// 本地示範: ./?lang=zh-Hant#002
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D，呼叫端先清除畫布並以 save/restore 包覆；w、h 為尺寸，u 為 [0, cycle) 內的秒數。畫面只取決於 u。
// 下方是元素程式碼；CELLS、常數與輔助函式位於 kit。

// @element 02

CELLS[1] = (k, w, h, u) => {
  stage(k);
  const N = 7, BW = 44, G = 20, x0 = (w - (N * BW + (N - 1) * G)) / 2, base = h - 30;
  const HS = [52, 84, 68, 112, 92, 136, 158];
  k.fillStyle = C.navy3; rr(k, x0 - 18, base, N * BW + (N - 1) * G + 36, 5, 2.5); k.fill();
  const lu = mod(u, CYC / 2);
  for (let i = 0; i < N; i++) {
    const t0 = 0.15 + i * 0.12, g = backOut(P(lu, t0, t0 + 0.5), 2.6) * (1 - eIn(P(lu, 1.78 + i * 0.045, 2.04 + i * 0.045)));
    if (g <= 0.001) continue;
    const hh = HS[i] * g, x = x0 + i * (BW + G);
    k.fillStyle = PAL[i % 4]; rr(k, x, base - hh, BW, hh, [12, 12, 3, 3]); k.fill();

    const d = spring(lu - t0 - 0.34, 16, 7) * (1 - eIn(P(lu, 1.7 + i * 0.045, 1.9 + i * 0.045)));
    if (d > 0.01) { k.fillStyle = '#fff'; k.beginPath(); k.arc(x + BW / 2, base - hh - 16, 6.5 * d, 0, 7); k.fill(); }
  }
};
