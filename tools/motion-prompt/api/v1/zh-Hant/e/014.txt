// 動態拖尾（Motion trail）
// 分類組: motion (動態圖像與動畫效果) · element 14 · 分類: 動態修飾
// 示範內容: 球沿八字形移動，後方留下逐漸淡去的副本軌跡。
// 動作／視覺規格:
//   - 球（珊瑚色、半徑 25、白色中心半徑 11）每 4.7 s 畫一個八字：x = 196·sin(a), y = 62·sin(2a)，a 以等速增加。
//   - 15 個副本沿路徑向後排列，彼此相差 0.105 rad（約 0.079 s）。
//   - 副本半徑由 22→6，不透明度為 0.9 × (1 − i/15) + 0.05；越舊的副本越小、越淡。
//   - 副本由新至舊依序呈現珊瑚→黃→藍。
//   - 先畫最舊的副本，最後才畫球本身。
// 使用情境: 適合用飛行物體、游標或光點表達速度與移動路徑。
// 關鍵字: trail, motion-trail, echo, afterimage, ghost, path, speed
// 尺寸: 539x208 px · 週期: 4.7 s
// 需要: kit motion -> ./api/v1/kit/zh-Hant/motion.js
// 本地示範: ./?lang=zh-Hant#014
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D，呼叫端先清除畫布並以 save/restore 包覆；w、h 為尺寸，u 為 [0, cycle) 內的秒數。畫面只取決於 u。
// 下方是元素程式碼；CELLS、常數與輔助函式位於 kit。

// @element 14

CELLS[13] = (k, w, h, u) => {
  stage(k);
  const X = w / 2, Y = h / 2 + 2, A = 196, B = 62, th = u / CYC * Math.PI * 2;
  const pt = a => [X + A * Math.sin(a), Y + B * Math.sin(2 * a)];
  const N = 15;
  for (let i = N; i >= 1; i--) {
    const [x, y] = pt(th - i * 0.105), f = i / N;
    k.globalAlpha = 0.9 * (1 - f) + 0.05; k.fillStyle = f < 0.5 ? mix(C.coral, C.yel, f * 2) : mix(C.yel, C.blue, (f - 0.5) * 2);
    k.beginPath(); k.arc(x, y, lerp(22, 6, f), 0, 7); k.fill();
  }
  k.globalAlpha = 1; const [x, y] = pt(th);
  k.fillStyle = C.coral; k.beginPath(); k.arc(x, y, 25, 0, 7); k.fill();
  k.fillStyle = '#fff'; k.beginPath(); k.arc(x, y, 11, 0, 7); k.fill();
};
