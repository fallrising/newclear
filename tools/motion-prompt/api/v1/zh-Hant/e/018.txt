// 圖形轉場（Shape transition）
// 分類組: motion (動態圖像與動畫效果) · element 18 · 分類: 轉場與重點效果
// 示範內容: 彩色色帶掃過並遮住畫面，離開時顯露下一個場景。
// 動作／視覺規格:
//   - 白天→夜晚→白天，各段 2.35 s。四條水平色帶（高 52；藍、珊瑚、黃、薄荷綠），兩端斜切 80 px。
//   - 覆蓋：0.38–1.0 s，各色帶前緣從畫面一側移至另一側；各延遲階段長度的 14%，並在階段長度的 58% 內採三次方 ease-in-out 移動。
//   - 1.0 s 完全遮住畫面時，底下場景換成下一幕。
//   - 揭開：1.12–1.75 s，各色帶後緣以相同延遲與曲線跟進，露出下一幕。
//   - 方向：白天→夜晚由左至右，夜晚→白天則左右鏡像反轉。
// 使用情境: 適合用品牌色圖形切換章節或話題。
// 關鍵字: shape-transition, transition, wipe, bands, stripes, cover, scene-change
// 尺寸: 539x208 px · 週期: 4.7 s
// 需要: kit motion -> ./api/v1/kit/zh-Hant/motion.js
// 本地示範: ./?lang=zh-Hant#018
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D，呼叫端先清除畫布並以 save/restore 包覆；w、h 為尺寸，u 為 [0, cycle) 內的秒數。畫面只取決於 u。
// 下方是元素程式碼；CELLS、常數與輔助函式位於 kit。

// @element 18

CELLS[17] = abCell((k, w, h, u, a, b) => {
  const p1 = P(u, TA - 0.12, TA + 0.5), p2 = P(u, TA + 0.62, TA + 1.25), N = 4, sk = 80;
  SCR[p1 >= 1 ? b : a](k, w, h);
  const cols = [C.blue, C.coral, C.yel, C.mint];
  for (let i = 0; i < N; i++) {
    const d = i * 0.14, e1 = eIO(clamp((p1 - d) / (1 - 0.42))), e2 = eIO(clamp((p2 - d) / (1 - 0.42)));
    if (e1 <= 0 || e2 >= 1) continue;
    const bh = h / N + 1, y = i * h / N, span = w + sk + 20, dir = a === 0 ? 1 : -1;
    let xa = -sk - 10 + e2 * span, xb = -sk - 10 + e1 * span;
    k.save(); if (dir < 0) { k.translate(w, 0); k.scale(-1, 1); }
    k.fillStyle = cols[i]; k.beginPath(); k.moveTo(xa + sk, y); k.lineTo(xb + sk, y); k.lineTo(xb, y + bh); k.lineTo(xa, y + bh); k.closePath(); k.fill(); k.restore();
  }
});
