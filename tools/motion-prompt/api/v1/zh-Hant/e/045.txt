// 動作／視覺規格:
//   - 接點：1.7 s 至 2.8 s，共 1.1 s；邊界從左側外（x = -4 px）移至右側外（x = 543 px）。
//   - 緩動：三次方 ease-in-out（約 cubic-bezier(0.65,0,0.35,1)）。
//   - 邊界左側顯示下一幕、右側顯示前一幕；兩幕本身都不移動，是裁切揭露而非推動畫面。
//   - 邊界：4 px 藍色垂直線，只有移動期間可見。
//   - 攝影機與主體在兩幕中都固定。
//   - 每 4.7 s 交替 A→B、B→A（往返 9.4 s）；右上角顯示「A→B」或「B→A」。
// 使用情境: 適合明確切換場景或話題，或在對照的兩側之間切換。
// 關鍵字: wipe, linear wipe, reveal, clip, edge, left to right, transition
// 尺寸: 539x208 px · 週期: 9.4 s
// 需要: kit camera -> ./api/v1/kit/zh-Hant/camera.js
// 本地示範: ./?lang=zh-Hant#045
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D，呼叫端先清除畫布並以 save/restore 包覆；w、h 為尺寸，u 為 [0, cycle) 內的秒數。畫面只取決於 u。
// 下方是元素程式碼；CELLS、常數與輔助函式位於 kit。

// @element 21

CELLS[20] = shot(abCell((k, w, h, u, A, B) => {
  const p = eIO(P(u, 1.7, 2.8)), ex = p * (w + 8) - 4;
  A(k, w, h);
  if (p > 0) { k.save(); k.beginPath(); k.rect(0, 0, ex, h); k.clip(); k.fillStyle = '#fff'; k.fillRect(0, 0, w, h); B(k, w, h); k.restore(); }
  if (p > 0 && p < 1) { k.fillStyle = C.blue; k.fillRect(ex - 2, 0, 4, h); }
}));
