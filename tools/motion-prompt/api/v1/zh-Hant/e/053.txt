// 下拉選單（Dropdown (select)）
// 分類組: ui (UI 元件) · element 05 · 分類: 展開與浮層
// 示範內容: 點選欄位後，下方顯示四個選項；選擇名古屋後，清單關閉並填入欄位。
// 動作／視覺規格:
//   - 542×191 卡片上每 4.5 s 循環。260×38 欄位（圓角半徑 8）顯示提示文字與箭頭；游標於 1.0 s 點擊。
//   - 開啟量：於 1.0–1.25 s 以 easeOutCubic 從 0→1。白色 260×112 清單（陰影模糊 18）下移 6 px 並淡入；箭頭旋轉 180°，欄位邊框變為 2 px 藍線。
//   - 四個選項相隔 26 px：東京、大阪、名古屋、福岡。游標下移時，大阪從 1.55 s、名古屋從 1.95 s 起以淡藍底反白。
//   - 2.3 s 點選名古屋：清單於 2.3–2.45 s 關閉（1 − easeInOutCubic）；提示文字於 2.2–2.3 s 淡出，名古屋於 2.32–2.42 s 淡入。
//   - 重設：名古屋於 3.9–4.08 s 淡出，提示文字於 4.1–4.3 s 恢復，兩段文字不重疊。
// 使用情境: 適合在表單中，從固定清單選擇一個值。
// 關鍵字: dropdown, dropdown-menu, select, picker, combobox, options, form
// 尺寸: 542x191 px · 週期: 4.5 s
// 需要: kit ui -> ./api/v1/kit/zh-Hant/ui.js
// 本地示範: ./?lang=zh-Hant#053
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D，呼叫端先清除畫布並以 save/restore 包覆；w、h 為尺寸，u 為 [0, cycle) 內的秒數。畫面只取決於 u。
// 下方是元素程式碼；CELLS、常數與輔助函式位於 kit。

// @element 05

CELLS[4] = ui((w, h, t) => {
  bar(24, 14, 90, C.line2, 10);
  const open = eo(seg(t, 1.0, 1.25)) * (1 - eio(seg(t, 2.3, 2.45)));
  const chosen = seg(t, 2.32, 2.42) * (1 - seg(t, 3.9, 4.08));
  const ph = (1 - seg(t, 2.2, 2.3)) + seg(t, 4.1, 4.3);
  strokeRR(24, 34, 260, 38, 8, open > .1 || (chosen > 0 && t < 2.9) ? C.blue : C.line2, open > .1 ? 2 : 1.5);
  if (ph > 0) { X.globalAlpha = clamp(ph); text("請選擇", 40, 53, 15, '#9a9896', 500); X.globalAlpha = 1; }
  if (chosen > 0) { X.globalAlpha = chosen; text("名古屋", 40, 53, 16, C.ink, 600); X.globalAlpha = 1; }
  chevron(262, 52, 5, Math.PI * open, '#777');
  fillRR(310, 34, 208, 70, 8, C.block); bar(310, 120, 208); bar(310, 138, 160); bar(24, 150, 200); bar(24, 168, 150);
  if (open > 0) {
    X.save(); X.globalAlpha = open; const oy = 78 - 6 * (1 - open);
    shadowed(() => fillRR(24, oy, 260, 112, 8, '#fff'), 18, 6, .18); strokeRR(24, oy, 260, 112, 8, C.border, 1);
    const hi = seg(t, 1.55, 1.75) > 0 ? (t < 1.95 ? 1 : 2) : -1;
    ["東京", "大阪", "名古屋", "福岡"].forEach((s, i) => { const yy = oy + 4 + i * 26; if (i === hi + (hi === 1 ? 0 : 0) && hi >= 0) fillRR(28, yy, 252, 26, 5, C.lb); text(s, 40, yy + 13, 15, i === hi ? C.blue : '#444', i === hi ? 600 : 500); });
    X.restore();
  }
  pointer([[.3, w + 30, h + 30], [.95, 200, 56], [1.25, 200, 56], [1.6, 170, 121], [2.0, 170, 147], [2.45, 170, 147], [3.2, 170, 147], [4.2, w + 30, h + 30]], [1.0, 2.3], t);
});
