// 浮動面板（Popover）
// 分類組: ui (UI 元件) · element 07 · 分類: 展開與浮層
// 示範內容: 點選詳情後，右側開啟帶指示尖角的白色小面板；點選內部確定關閉。
// 動作／視覺規格:
//   - 542×191 卡片上每 4.5 s 循環。藍色外框詳情按鈕（108×36）位於 (24, 112)，游標於 1.0 s 點擊。
//   - 顯示量 a：於 1.0–1.28 s 以 easeOutCubic 從 0→1；不透明度 a，從右側滑入 6 px。
//   - 面板：白色 300×120、圓角半徑 10，位於 (156, 62)，邊框 1 px，陰影模糊 22、y 偏移 6；左緣寬 10 的三角形（y 122–138）指向按鈕。
//   - 內容：粗標題列、兩行文字、外框 70×26 按鈕與藍色 72×26 確定按鈕。
//   - 關閉：游標於 3.0 s 前移至確定 (402, 157)，3.3 s 點擊；a 乘上 (1 − 3.3–3.5 s 期間的 easeInOutCubic)。
// 使用情境: 適合緊鄰觸發按鈕的簡短設定或確認操作。
// 關鍵字: popover, popup, callout, bubble, flyout, overlay
// 尺寸: 542x191 px · 週期: 4.5 s
// 需要: kit ui -> ./api/v1/kit/zh-Hant/ui.js
// 本地示範: ./?lang=zh-Hant#055
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D，呼叫端先清除畫布並以 save/restore 包覆；w、h 為尺寸，u 為 [0, cycle) 內的秒數。畫面只取決於 u。
// 下方是元素程式碼；CELLS、常數與輔助函式位於 kit。

// @element 07

CELLS[6] = ui((w, h, t) => {
  pageLines(24, 16, 494);
  strokeRR(24, 112, 108, 36, 8, C.blue, 2); text("詳情", 78, 130, 15, C.blue, 600, 'center');
  fillRR(24, 160, 494, 22, 6, C.block);
  const a = eo(seg(t, 1.0, 1.28)) * (1 - eio(seg(t, 3.3, 3.5)));
  if (a > 0) {
    X.save(); X.globalAlpha = a; X.translate(6 * (1 - a), 0);
    const px = 156, py = 62, pw = 300, ph = 120;
    shadowed(() => { fillRR(px, py, pw, ph, 10, '#fff'); }, 22, 6, .2); strokeRR(px, py, pw, ph, 10, C.border, 1);
    X.beginPath(); X.moveTo(px + 1, 122); X.lineTo(px - 10, 130); X.lineTo(px + 1, 138); X.fillStyle = '#fff'; X.fill();
    X.beginPath(); X.moveTo(px, 122); X.lineTo(px - 10, 130); X.lineTo(px, 138); X.strokeStyle = C.border; X.lineWidth = 1; X.stroke();
    bar(px + 18, py + 18, 130, C.ink, 11); bar(px + 18, py + 42, 250); bar(px + 18, py + 60, 200);
    strokeRR(px + 130, py + 82, 70, 26, 6, C.line2, 1.5); fillRR(px + 210, py + 82, 72, 26, 6, C.blue); text("確定", px + 246, py + 95, 14, '#fff', 700, 'center', NUMF);
    X.restore();
  }
  pointer([[.3, w + 30, h + 30], [.95, 84, 134], [1.6, 84, 134], [3.0, 402, 157], [3.5, 402, 157], [4.2, w + 30, h + 30]], [1.0, 3.3], t);
});
