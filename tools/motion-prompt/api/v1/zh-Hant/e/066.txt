  const kx = x, kw = tw("課程", 17, 700);
  const hoverK = t > .9 && t < 1.2;
  text("課程", x, y, 17, up > .5 ? C.ink : C.blue, up > .5 ? 700 : 600); if (hoverK) { X.fillStyle = C.blue; X.fillRect(x, y + 12, kw, 2); }
  x += kw + 10;
  X.globalAlpha = 1 - up; text('›', x, y, 17, '#aaa', 500); text("網頁製作", x + 18, y, 17, C.ink, 700); X.globalAlpha = 1;
  X.fillStyle = C.line; X.fillRect(24, 46, 494, 1.5);

});
