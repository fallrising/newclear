// 視差捲動（Parallax scrolling）
// 分類組: lp (網頁與登陸頁效果) · element 01 · 分類: 捲動效果
// 示範內容: 頁面向下捲動時，標題正常向上移動，天空與太陽、遠山及近山則移動得比較慢。
// 動作／視覺規格:
//   - 瀏覽器視窗的可視區域為 511 × 168。捲動：0.3–1.9 s 從 0→120 px，2.9–4.3 s 從 120→0（cubic ease-in-out）。
//   - 各圖層位移為捲動量 × 係數：天空與太陽 0.15、遠山 0.4、近山 0.7。
//   - 標題（28 px）、按鈕與下一個區塊的係數為 1.0（正常捲動）。
//   - 下一個白色區塊從主視覺底部（168 px）開始，滑過移動較慢的背景。
//   - 右側捲軸滑塊隨捲動移動。
// 使用情境: 適合讓主視覺或區塊照片呈現景深層次。
// 關鍵字: parallax, parallax-scrolling, scroll, depth, layers, hero, background
// 尺寸: 539x208 px · 週期: 4.8 s
// 需要: kit lp -> ./api/v1/kit/zh-Hant/lp.js
// 本地示範: ./?lang=zh-Hant#073
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D，呼叫端先清除畫布並以 save/restore 包覆；w、h 為尺寸，u 為 [0, cycle) 內的秒數。畫面只取決於 u。
// 下方是元素程式碼；CELLS、常數與輔助函式位於 kit。

// @element 01

CELLS[0] = (k, w, h, u) => {
  frame(k);
  const s = steps([[0.3, 0], [1.9, 1], [2.9, 1], [4.3, 0]], u), y = s * 120;
  view(k, () => {
    const g = k.createLinearGradient(0, 0, 0, 200); g.addColorStop(0, '#9DB6FB'); g.addColorStop(1, '#E6ECFF');
    k.save(); k.translate(0, -y * 0.15);
    k.fillStyle = g; k.fillRect(0, 0, VW, 260); k.fillStyle = C.acc; k.beginPath(); k.arc(392, 54, 25, 0, 7); k.fill(); k.restore();
    k.save(); k.translate(0, -y * 0.4); mount(k, VW, 190, M1, '#7D9AF4', 0); k.restore();
    k.save(); k.translate(0, -y * 0.7); mount(k, VW, 205, M2, '#4B6BDB', 1); k.restore();
    k.save(); k.translate(0, -y);
    txt(k, "一起走進山林。", 34, 64, `800 28px ${JP}`, '#fff', 'left', 'alphabetic', '1px');
    bars(k, 34, 78, 160, 2, 'rgba(255,255,255,.85)');
    btn(k, 34, 106, 92, 26, "預約");
    k.fillStyle = '#fff'; k.fillRect(0, VH, VW, 200);
    txt(k, "熱門課程", 34, VH + 34, `800 16px ${JP}`, C.ink);
    for (let i = 0; i < 3; i++) { pic(k, 34 + i * 152, VH + 50, 136, 58, i, 6); bars(k, 34 + i * 152, VH + 116, 110, 1, C.faint); }
    k.restore();
    sbar(k, s * 0.35, 0.4);
  });
};
