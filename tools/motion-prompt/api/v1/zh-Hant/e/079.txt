// 閱讀進度條（Scroll progress bar）
// 分類組: lp (網頁與登陸頁效果) · element 07 · 分類: 捲動效果
// 示範內容: 文章逐段向下閱讀時，頂部線條從左側延伸，到頁尾時填滿整條。
// 動作／視覺規格:
//   - 文章以四次 110 px 的步進，從 0→430 px；每次移動 0.45 s，接著停頓 0.4 s（cubic ease-in-out）。
//   - 3.75–4.55 s 從 430→0 返回。
//   - 可視區域頂部有 6 px 長條，軌道為淺灰色，填色為橘色。
//   - 填滿寬度 = 511 px × scroll / 430，直接對應捲動比例（如 CSS animation-timeline: scroll()）。
// 使用情境: 適合長篇文章，讓讀者知道還剩多少內容。
// 關鍵字: progress-bar, reading-progress, scroll-progress, indicator, scroll-driven, article
// 尺寸: 539x208 px · 週期: 4.8 s
// 需要: kit lp -> ./api/v1/kit/zh-Hant/lp.js
// 本地示範: ./?lang=zh-Hant#079
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D，呼叫端先清除畫布並以 save/restore 包覆；w、h 為尺寸，u 為 [0, cycle) 內的秒數。畫面只取決於 u。
// 下方是元素程式碼；CELLS、常數與輔助函式位於 kit。

// @element 07

CELLS[6] = (k, w, h, u) => {
  frame(k);
  const MAX = 430;
  const y = steps([[0.3, 0], [0.75, 110], [1.15, 110], [1.6, 220], [2.0, 220], [2.45, 330], [2.85, 330], [3.3, MAX], [3.75, MAX], [4.55, 0]], u);
  view(k, () => {
    k.save(); k.translate(0, -y);
    txt(k, "網頁設計入門", 70, 44, `800 20px ${JP}`, C.ink); bars(k, 70, 56, 120, 1, C.faint, 5);
    pic(k, 70, 74, 371, 112, 0, 8);
    for (let b = 0; b < 4; b++) { const y0 = 204 + b * 82; k.fillStyle = C.ink; rr(k, 70, y0, 140, 9, 4.5); k.fill(); bars(k, 70, y0 + 22, 371, 4, C.faint, 6, 12); }
    k.restore();
    k.fillStyle = C.faint; k.fillRect(0, 0, VW, 6);
    k.fillStyle = C.acc; k.fillRect(0, 0, VW * (y / MAX), 6);
    sbar(k, y / MAX, 0.3);
  });
};
