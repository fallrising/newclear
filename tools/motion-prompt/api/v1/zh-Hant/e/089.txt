// 分類組: lp (網頁與登陸頁效果) · element 17 · 分類: 載入、導覽與循環動態
// 示範內容: 深藍畫面中，標誌下的數字從 0% 增至 100%；接著畫面向上移走，露出逐漸淡入的頁面。
// 動作／視覺規格:
//   - 深藍畫面填滿可視區域，包含標誌、28 px 數字與 180 × 4 的長條。
//   - 0.25–2.1 s 從 0→100%（cubic ease-out，四捨五入為整數），長條以相同速率填入橘色。
//   - 2.3–2.85 s 整個畫面向上移動 168 px 並離開（cubic ease-in-out）。
//   - 2.35–2.95 s 頁面標題與照片由不透明度 0→1，並從下方 14 px 上升（cubic ease-out）。
//   - 4.25–4.75 s 畫面從頂部回到原位，顯示 0%，開始下一輪載入。
// 使用情境: 適合圖片或影片較多的網站，讓等待載入的時間呈現品牌特色。
// 關鍵字: preloader, loading-screen, loader, percentage, intro, splash, curtain
// 尺寸: 539x208 px · 週期: 4.8 s
// 需要: kit lp -> ./api/v1/kit/zh-Hant/lp.js
// 本地示範: ./?lang=zh-Hant#089
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D，呼叫端先清除畫布並以 save/restore 包覆；w、h 為尺寸，u 為 [0, cycle) 內的秒數。畫面只取決於 u。
// 下方是元素程式碼；CELLS、常數與輔助函式位於 kit。

// @element 17

CELLS[16] = (k, w, h, u) => {
  frame(k);
  const n = Math.round(100 * eOut(P(u, 0.25, 2.1)));
  const cy = u < 4.25 ? -VH * eIO(P(u, 2.3, 2.85)) : -VH * (1 - eIO(P(u, 4.25, 4.75)));
  const show = eOut(P(u, 2.35, 2.95));
  view(k, () => {
    header(k, 0);
    k.save(); k.globalAlpha = show; k.translate(0, 14 * (1 - show));
    txt(k, "歡迎。", 34, 82, `800 26px ${JP}`, C.ink); bars(k, 34, 98, 200, 3); btn(k, 34, 136, 96, 24, "開始");
    pic(k, 270, 46, 214, 112, 0, 8); k.restore();
    if (cy > -VH) {
      k.save(); k.translate(0, cy); k.fillStyle = C.ink; k.fillRect(0, 0, VW, VH);
      logo(k, VW / 2 - 40, 56, '#fff');
      txt(k, (u >= 4.25 ? 0 : n) + '%', VW / 2, 98, `800 28px ${EN}`, '#fff', 'center', 'middle');
      k.fillStyle = 'rgba(255,255,255,.15)'; rr(k, VW / 2 - 90, 122, 180, 4, 2); k.fill();
      k.fillStyle = C.acc; rr(k, VW / 2 - 90, 122, 180 * (u >= 4.25 ? 0 : n / 100), 4, 2); k.fill();
      k.restore();
    }
  });
};
