// 分類組: lp (網頁與登陸頁效果) · element 18 · 分類: 載入、導覽與循環動態
// 示範內容: 灰色占位區捲入畫面時，中央出現旋轉指示器，圖片載入後淡入；最上排一開始就已載入。
// 動作／視覺規格:
//   - 3 × 3 網格的占位區各為 151 × 92，第一排一開始即顯示；捲動：0.35–1.1 s 從 0→110 px，1.7–2.5 s 從 110→225 px，3.5–4.5 s 從 225→0（cubic ease-in-out）。
//   - 占位區頂緣越過可視區域底部（168 px）時開始載入，各欄比左側一欄晚 0.12 s。
//   - 占位區中央的旋轉指示器持續 0.45 s，半徑 11，每 0.8 s 轉一圈。
//   - 接著圖片在 0.35 s 內從不透明度 0→1（cubic ease-out）；尚未載入的占位區顯示圖片圖示。
//   - 從可視區域底部離開的占位區會在畫面外重設。
// 使用情境: 適合圖片較多的頁面，讓首屏更快載入。
// 關鍵字: lazy-loading, lazy-load, images, placeholder, performance, spinner, gallery
// 尺寸: 539x208 px · 週期: 4.8 s
// 需要: kit lp -> ./api/v1/kit/zh-Hant/lp.js
// 本地示範: ./?lang=zh-Hant#090
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D，呼叫端先清除畫布並以 save/restore 包覆；w、h 為尺寸，u 為 [0, cycle) 內的秒數。畫面只取決於 u。
// 下方是元素程式碼；CELLS、常數與輔助函式位於 kit。

// @element 18

CELLS[17] = (k, w, h, u) => {
  frame(k);
  const SY = t => steps([[0.35, 0], [1.1, 110], [1.7, 110], [2.5, 225], [3.5, 225], [4.5, 0]], t), y = SY(u);
  const COLX = [15, 180, 345], IW = 151, IH = 92, ROWS = [12, 168, 276], WAIT = 0.45, FADE = 0.35;
  view(k, () => {
    k.save(); k.translate(0, -y);
    txt(k, "圖庫", 15, 140, `800 18px ${JP}`, C.ink); bars(k, 120, 128, 200, 1);
    ROWS.forEach((top, r) => COLX.forEach((x, c) => {
      const t0 = r === 0 ? -1e9 : enteredAt(SY, top, VH, u), age = t0 === null ? -1 : u - t0 - c * 0.12;
      k.fillStyle = C.skel; rr(k, x, top, IW, IH, 6); k.fill();
      if (age < WAIT + FADE) {
        k.save(); k.translate(x + IW / 2, top + IH / 2); k.strokeStyle = '#B9C1CF'; k.lineWidth = 2; k.lineJoin = 'round';
        if (age >= 0 && age < WAIT) spinner(k, 0, 0, 11, u, C.blue);
        else { rr(k, -15, -11, 30, 22, 3); k.stroke(); k.beginPath(); k.moveTo(-11, 7); k.lineTo(-3, -1); k.lineTo(3, 5); k.lineTo(7, 1); k.lineTo(11, 6); k.stroke(); }
        k.restore();
      }
      if (age >= WAIT) { k.save(); k.globalAlpha = eOut(P(age, WAIT, WAIT + FADE)); pic(k, x, top, IW, IH, (r + c) % 3, 6); k.restore(); }
    }));
    k.restore();
    sbar(k, y / 225, 0.45);
  });
};
