// 無限捲動（Infinite scroll）
// 分類組: lp (網頁與登陸頁效果) · element 19 · 分類: 載入、導覽與循環動態
// 示範內容: 捲至清單末端會出現旋轉指示器，接著新增三筆項目，讓閱讀繼續。
// 動作／視覺規格:
//   - 清單項目為 331 × 44，圓角半徑 8，間隔 52；最後一項下方有半徑 10、每 0.8 s 轉一圈的旋轉指示器。
//   - 捲動：0.25–0.85 s 從 38→116 px，1.15–1.75 s 從 116→194 px（cubic ease-in-out），停止時清單末端可見。
//   - 2.6 s 新增三筆項目：各自在 0.35 s 內由不透明度 0→1，並從下方 10 px 上升（cubic ease-out）；旋轉指示器移至新的末端。
//   - 每輪推進三筆項目（156 px）；內容每三筆重複，因此週期結尾與開頭外觀一致。
// 使用情境: 適合動態消息與新聞清單，讓使用者持續閱讀而不用換頁。
// 關鍵字: infinite-scroll, endless-scroll, load-more, feed, pagination, spinner, list
// 尺寸: 539x208 px · 週期: 4.8 s
// 需要: kit lp -> ./api/v1/kit/zh-Hant/lp.js
// 本地示範: ./?lang=zh-Hant#091
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D，呼叫端先清除畫布並以 save/restore 包覆；w、h 為尺寸，u 為 [0, cycle) 內的秒數。畫面只取決於 u。
// 下方是元素程式碼；CELLS、常數與輔助函式位於 kit。

// @element 19

CELLS[18] = (k, w, h, u) => {
  frame(k);

  const PITCH = 52, BATCH = 3, L0 = 6, TL = 2.6;
  const y = steps([[0.25, 38], [0.85, 116], [1.15, 116], [1.75, 194], [4.8, 194]], u);
  const n = u < TL ? L0 : L0 + BATCH, end = 10 + n * PITCH;
  view(k, () => {
    k.fillStyle = C.soft; k.fillRect(0, 0, VW, VH);
    k.save(); k.translate(0, -y);
    for (let i = 0; i < n; i++) {
      const top = 10 + i * PITCH, v = i % 3, fresh = i >= L0 ? eOut(P(u, TL, TL + 0.35)) : 1;
      if (top - y > VH || top + 44 - y < 0) continue;
      k.save(); k.globalAlpha = fresh; k.translate(0, 10 * (1 - fresh));
      k.fillStyle = '#fff'; rr(k, 90, top, 331, 44, 8); k.fill(); pic(k, 98, top + 6, 48, 32, v, 5);
      bars(k, 158, top + 12, [180, 140, 200][v], 1, C.ink, 7); bars(k, 158, top + 26, [120, 150, 100][v], 1, C.faint, 6);
      k.restore();
    }
    spinner(k, VW / 2, end + 18, 10, u);
    k.restore();
  });
};
