// 頁面轉場（Page transition）
// 分類組: lp (網頁與登陸頁效果) · element 20 · 分類: 載入、導覽與循環動態
// 示範內容: 點擊作品清單卡片後，橘色面板從左側掃入；網址切換後，面板在詳情頁向右退出。「返回清單」以相同方式返回。
// 動作／視覺規格:
//   - 0.8 s 點擊卡片；0.9–1.3 s 橘色全畫面面板從左側進入並完全遮住內容，中央有標誌（cubic ease-in-out）。
//   - 1.3 s 完全遮住時，頁面與網址列切換（/works → /works/02）。
//   - 1.4–1.85 s 面板向右退出（cubic ease-in-out）。
//   - 3.0 s 點擊「← 返回清單」：3.1–3.5 s 面板進入，3.5 s 切換頁面，3.6–4.05 s 面板退出。
//   - 面板覆蓋整個可視區域，包含頁首。
// 使用情境: 適合品牌網站與作品集，用動畫串接頁面。
// 關鍵字: page-transition, transition, view-transitions, navigation, wipe, barba, spa
// 尺寸: 539x208 px · 週期: 4.8 s
// 需要: kit lp -> ./api/v1/kit/zh-Hant/lp.js
// 本地示範: ./?lang=zh-Hant#092
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D，呼叫端先清除畫布並以 save/restore 包覆；w、h 為尺寸，u 為 [0, cycle) 內的秒數。畫面只取決於 u。
// 下方是元素程式碼；CELLS、常數與輔助函式位於 kit。

// @element 20

CELLS[19] = (k, w, h, u) => {
  frame(k, u >= 1.3 && u < 3.5 ? 'example.com/works/02' : 'example.com/works');
  const CLK = [0.8, 3.0];
  const cover = t0 => eIO(P(u, t0, t0 + 0.4)), uncover = t0 => eIO(P(u, t0, t0 + 0.45));

  const xs = u < 1.4 ? -VW * (1 - cover(0.9)) : u < 3.1 ? VW * uncover(1.4) : u < 3.6 ? -VW * (1 - cover(3.1)) : VW * uncover(3.6);
  const pageB = u >= 1.3 && u < 3.5;
  const [cx, cy] = path([[0, 420, 150], [0.65, 255, 100], [2.6, 255, 100], [2.85, 46, 50], [3.3, 46, 50], [4.6, 420, 150], [4.8, 420, 150]], u);
  view(k, () => {
    header(k, 0);
    if (!pageB) {
      txt(k, "作品", 34, 62, `800 18px ${EN}`, C.ink, 'left', 'alphabetic', '2px');
      for (let i = 0; i < 3; i++) { const x = 34 + i * 152; pic(k, x, 74, 136, 60, i, 6); bars(k, x, 142, 100, 1, C.ink, 7); }
    } else {
      txt(k, "← 返回清單", 34, 55, `700 12px ${JP}`, C.sub, 'left', 'middle');
      pic(k, 34, 68, 240, 90, 1, 8);
      txt(k, "專案 02", 296, 96, `800 22px ${EN}`, C.ink); bars(k, 296, 110, 180, 3);
    }
    if (xs > -VW && xs < VW) { k.fillStyle = C.acc; k.fillRect(xs, 0, VW, VH); logo(k, xs + VW / 2 - 40, VH / 2, '#fff'); }
    click(k, cx, cy, u, CLK);
  });
};
