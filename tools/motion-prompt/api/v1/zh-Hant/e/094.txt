// 示範內容: 深藍色帶狀區域內，大型服務名稱以橘色圓點分隔，不間斷地向左移動。
// 動作／視覺規格:
//   - 帶狀區域高 58、深藍色，填滿可視寬度；文字為 30 px 特粗白字，字距 1 px。
//   - 一組包含四個詞（WEB DESIGN / BRANDING / MOTION / UI/UX），彼此間隔 64 px，每個間隙有半徑 6 的橘色圓點。
//   - 每個週期（4.8 s）以等速向左移動一組文字的寬度，此例約 881 px；量測整組寬度並依該寬度循環位移，接縫不會顯現。
//   - 重複整組內容，直到填滿可視區域寬度。
// 使用情境: 適合有活力地呈現服務名稱、客戶標誌或標語。
// 關鍵字: marquee, ticker, infinite-loop, scrolling-text, logo-strip, band, loop
// 尺寸: 539x208 px · 週期: 4.8 s
// 需要: kit lp -> ./api/v1/kit/zh-Hant/lp.js
// 本地示範: ./?lang=zh-Hant#094
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D，呼叫端先清除畫布並以 save/restore 包覆；w、h 為尺寸，u 為 [0, cycle) 內的秒數。畫面只取決於 u。
// 下方是元素程式碼；CELLS、常數與輔助函式位於 kit。

// @element 22

CELLS[21] = (k, w, h, u) => {
  frame(k);
  const WORDS = ['WEB DESIGN', 'BRANDING', 'MOTION', 'UI/UX'], F = `800 30px ${EN}`, GAPW = 64;
  view(k, () => {
    bars(k, 34, 20, 180, 1, C.ink, 10); bars(k, 34, 38, 300, 1);
    k.fillStyle = C.ink; k.fillRect(0, 58, VW, 58);
    k.font = F; k.letterSpacing = '1px'; const ws = WORDS.map(s => k.measureText(s).width); k.letterSpacing = '0px';
    const unit = ws.reduce((a, b) => a + b + GAPW, 0);
    let x = -mod(u / CYC, 1) * unit;
    for (let rep = 0; x < VW; rep++) WORDS.forEach((s, i) => {
      txt(k, s, x, 88, F, '#fff', 'left', 'middle', '1px');
      k.fillStyle = C.acc; k.beginPath(); k.arc(x + ws[i] + GAPW / 2, 88, 6, 0, 7); k.fill();
      x += ws[i] + GAPW;
    });
    for (let i = 0; i < 3; i++) { k.fillStyle = C.soft; rr(k, 34 + i * 152, 130, 136, 30, 6); k.fill(); bars(k, 46 + i * 152, 142, 90, 1, C.faint); }
  });
};
