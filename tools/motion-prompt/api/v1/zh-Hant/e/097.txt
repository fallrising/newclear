// 逆光（Backlighting）
// 分類組: photo (攝影效果、打光與風格) · element 01 · 分類: 打光與時段
// 示範內容: 太陽從畫面外移到人物頭部正後方；人物輪廓出現細亮邊，畫面覆上淡暖色薄霧，朝向鏡頭的一側變暗。
// 動作／視覺規格:
//   - 場景 1：地平線位於 y 150，前方為頭肩人物，頭部中心 x 270、y 78、半徑 29 px，沒有臉部或頭髮。亮部為平面琥珀色 #FFC75A，陰影為平面色 #33405A。
//   - 太陽半徑 20 px，從畫面左上方之外（x 40、y −40）移至頭部正後方（x 270、y 56）；疊加半徑 2.6× 與 1.7× 的平面光圈，色彩 #FFE3A3、不透明度為 35% 與 60%。
//   - 輪廓先以 7 px 線條描邊再填色，外側留下 3.5 px 的亮邊。
//   - 人物 #4D5A75 → #33405A；天空 #FAF9F7 → #FFF4DC；整個畫面疊上 22% 的 #FFE3A3。
//   - 時間：0.4 s 到 1.4 s 漸入，維持至 4.0 s，於 4.8 s 前退去；每輪 = 5 s，採 cubic ease-in-out，約為 cubic-bezier(0.65,0,0.35,1)。
// 使用情境: 適合通透發光的人像，或想營造沐浴在光線中感覺的影像。
// 關鍵字: backlighting, backlit, contre-jour, sun behind, haze, glow, halo
// 尺寸: 539x208 px · 週期: 5 s
// 需要: kit photo -> ./api/v1/kit/zh-Hant/photo.js
// 本地示範: ./?lang=zh-Hant#097
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D，呼叫端先清除畫布並以 save/restore 包覆；w、h 為尺寸，u 為 [0, cycle) 內的秒數。畫面只取決於 u。
// 下方是元素程式碼；CELLS、常數與輔助函式位於 kit。

// @element 01

CELLS[0] = cell((k, u) => {
  const a = env(u);
  const BACK = Object.assign({}, BASE, { sky: '#FFF4DC', sky2: '#FFEBC4', far: '#F6E2BE', ground: '#F7EAD3', bld: '#F7E9D2', winOff: '#EEDCBF', win: '#EEDCBF', tree: '#EEDCBF', edge: '#E3CFAE', post: '#CDB894', trunk: '#E3CFAE', bulb: '#E3CFAE' });
  stagePortrait(k, pal(BASE, BACK, a), { sun: [lerp(40, PX, a), lerp(-40, PY - 22, a), 20, 1] });

  k.save(); k.globalAlpha = a; bustPath(k, PX, PY, PR); k.strokeStyle = BASE.lit; k.lineWidth = 7; k.lineJoin = 'round'; k.stroke(); k.restore();
  bust(k, PX, PY, PR, mixc(BASE.person, BASE.shade, a));

  k.save(); k.globalAlpha = 0.22 * a; k.fillStyle = BASE.glow; k.fillRect(0, 0, CW, CH); k.restore();
});
