// 示範內容: 以明亮暮色天空為曝光基準後，人物、建築、路燈與樹木變成沒有細節的黑色形狀，只有天空保持明亮。
// 動作／視覺規格:
//   - 場景 1：地平線位於 y 150，前方為頭肩人物，頭部中心 x 270、y 78、半徑 29 px，沒有臉部或頭髮。
//   - 天空：頂部 #FAF9F7 → #FFEDC4，地平線附近 y 92–150 為 #FFCF73 色帶。
//   - 天空以外的物件變成 #2A2D36，遠山為 #4A4546，人物為 #22252D。
//   - 太陽半徑 20 px，位於 x 404，從 y 60 下沉至 y 128，一半被山丘遮住。
//   - 右上讀值：±0.0 EV → −2.0 EV。
//   - 時間：0.4 s 到 1.4 s 漸入，維持至 4.0 s，於 4.8 s 前退去；每輪 = 5 s，採 cubic ease-in-out，約為 cubic-bezier(0.65,0,0.35,1)。
// 使用情境: 適合以姿態與外形傳達畫面重點，搭配夕陽或明亮窗戶。
// 關鍵字: silhouette, backlit, sunset, contre-jour, black shape, underexposed
// 尺寸: 539x208 px · 週期: 5 s
// 需要: kit photo -> ./api/v1/kit/zh-Hant/photo.js
// 本地示範: ./?lang=zh-Hant#099
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D，呼叫端先清除畫布並以 save/restore 包覆；w、h 為尺寸，u 為 [0, cycle) 內的秒數。畫面只取決於 u。
// 下方是元素程式碼；CELLS、常數與輔助函式位於 kit。

// @element 03

CELLS[2] = cell((k, u) => {
  const a = env(u), INK = '#2A2D36';
  const SILH = mapPal(BASE, () => INK);
  Object.assign(SILH, { sky: '#FFEDC4', sky2: '#FFCF73', far: '#4A4546', sun: BASE.sun, glow: BASE.glow });
  stagePortrait(k, pal(BASE, SILH, a), { sun: [404, lerp(60, 128, a), 20, a] });
  bust(k, PX, PY, PR, mixc(BASE.person, '#22252D', a));
  hud(k, `${(-2 * a).toFixed(1).replace('-0.0', '±0.0')} EV`);
});
