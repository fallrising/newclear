// 分割光（Split lighting）
// 分類組: photo (攝影效果、打光與風格) · element 04 · 分類: 打光與時段
// 示範內容: 光源從接近正面的 20° 移至正側面 90°；頭部陰影從月牙狀擴大至恰好一半，沿中央分開臉部。
// 動作／視覺規格:
//   - 場景 1：地平線位於 y 150，前方為頭肩人物，頭部中心 x 270、y 78、半徑 29 px，沒有臉部或頭髮。背景轉暗，天空為 #E7E5E1。
//   - 光從畫面左側照射，角度 20° → 90°；右下為俯視圖，右上讀值為「90° · ↑0°」。
//   - 頭部：x = 270 + cos(angle)·√(29² − dy²) 的右側為陰影 #33405A，左側為亮部 #FFC75A；90° 時邊界是筆直的中心線。
//   - 身體半寬 56 px：x = 270 + cos(angle)·56 的右側為陰影，左側邊緣有一條亮部。
//   - 時間：0.4 s 到 1.4 s 漸入，維持至 4.0 s，於 4.8 s 前退去；每輪 = 5 s，採 cubic ease-in-out，約為 cubic-bezier(0.65,0,0.35,1)。
// 使用情境: 適合戲劇性、緊張或情緒強烈的人像。
// 關鍵字: split lighting, side light, portrait lighting, half face, dramatic, chiaroscuro
// 尺寸: 539x208 px · 週期: 5 s
// 需要: kit photo -> ./api/v1/kit/zh-Hant/photo.js
// 本地示範: ./?lang=zh-Hant#100
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D，呼叫端先清除畫布並以 save/restore 包覆；w、h 為尺寸，u 為 [0, cycle) 內的秒數。畫面只取決於 u。
// 下方是元素程式碼；CELLS、常數與輔助函式位於 kit。

// @element 04

CELLS[3] = cell((k, u) => {
  const a = env(u), az = lerp(0.35, Math.PI / 2, a);
  stagePortrait(k, pal(BASE, DIM, a));
  const lit = mixc(BASE.person, BASE.lit, a), sh = mixc(BASE.person, BASE.shade, a), W = torsoW(PR);
  bust(k, PX, PY, PR, BASE.person, kk => {
    kk.fillStyle = lit; kk.beginPath(); kk.arc(PX, PY, PR, 0, TAU); kk.fill();
    headShade(kk, PX, PY, PR, az, 1, sh);
    torso(kk, PX, PY, PR, PX - W, PX - Math.cos(az) * 0.4 * W - 0.4 * W, lit);
    torso(kk, PX, PY, PR, PX + Math.cos(az) * W, PX + W, sh);
  });
  plan(k, [[az, a]]);
  hud(k, `${Math.round(az * 180 / Math.PI)}° · ↑0°`);
});
