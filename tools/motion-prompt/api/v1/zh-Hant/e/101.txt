// 林布蘭光（Rembrandt lighting）
// 分類組: photo (攝影效果、打光與風格) · element 05 · 分類: 打光與時段
// 示範內容: 光源位於側面 45°、上方 45°；右半張臉落入陰影，只在陰影側臉頰留下小型倒三角形亮部。
// 動作／視覺規格:
//   - 場景 1：地平線位於 y 150，前方為頭肩人物，頭部中心 x 270、y 78、半徑 29 px，沒有臉部或頭髮。背景轉暗，天空為 #E7E5E1。
//   - 頭部：x = 270 + 0.04·29 的右側為陰影 #33405A，左側為亮部 #FFC75A。
//   - 亮三角形頂點為 (x 270+0.24R, y 78−0.12R)、(x 270+0.7R, y 78−0.12R)、(x 270+0.24R, y 78+0.5R)，R = 29；約寬 13 px、高 18 px，與亮側之間隔著約 6 px 寬的陰影帶。
//   - 身體：中心左側 −0.45×56 px 以外為亮部，右側 +0.3×56 px 以外為陰影；俯視圖光源在 45°，右上讀值為「45° · ↑45°」。
//   - 時間：0.4 s 到 1.4 s 漸入，維持至 4.0 s，於 4.8 s 前退去；每輪 = 5 s，採 cubic ease-in-out，約為 cubic-bezier(0.65,0,0.35,1)。
// 使用情境: 適合有繪畫感、立體雕塑感與古典氣質的人像。
// 關鍵字: rembrandt lighting, portrait lighting, triangle, 45 degree light, chiaroscuro, painterly
// 尺寸: 539x208 px · 週期: 5 s
// 需要: kit photo -> ./api/v1/kit/zh-Hant/photo.js
// 本地示範: ./?lang=zh-Hant#101
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D，呼叫端先清除畫布並以 save/restore 包覆；w、h 為尺寸，u 為 [0, cycle) 內的秒數。畫面只取決於 u。
// 下方是元素程式碼；CELLS、常數與輔助函式位於 kit。

// @element 05

CELLS[4] = cell((k, u) => {
  const a = env(u), R = PR, W = torsoW(R);
  stagePortrait(k, pal(BASE, DIM, a));
  const lit = mixc(BASE.person, BASE.lit, a), sh = mixc(BASE.person, BASE.shade, a);
  bust(k, PX, PY, R, BASE.person, kk => {
    kk.fillStyle = lit; kk.beginPath(); kk.arc(PX, PY, R, 0, TAU); kk.fill();
    kk.fillStyle = sh; kk.fillRect(PX + 0.04 * R, PY - R - 2, R * 2, R * 2 + 4);
    kk.beginPath(); kk.arc(PX, PY, R, 0, TAU); kk.clip();
    kk.fillStyle = lit; kk.beginPath();
    kk.moveTo(PX + 0.24 * R, PY - 0.12 * R); kk.lineTo(PX + 0.7 * R, PY - 0.12 * R); kk.lineTo(PX + 0.24 * R, PY + 0.5 * R); kk.closePath(); kk.fill();
  });
  bust(k, PX, PY, R, 'rgba(0,0,0,0)', kk => {
    kk.fillStyle = 'rgba(0,0,0,0)';
    torso(kk, PX, PY, R, PX - W, PX - 0.45 * W, lit);
    torso(kk, PX, PY, R, PX + 0.3 * W, PX + W, sh);
  });
  plan(k, [[Math.PI / 4, a]]);
  hud(k, '45° · ↑45°');
});
