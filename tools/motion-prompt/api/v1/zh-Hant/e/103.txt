// 黃金時刻（Golden hour）
// 分類組: photo (攝影效果、打光與風格) · element 07 · 分類: 打光與時段
// 示範內容: 太陽降至地平線上方；天空與地面轉橘，人物左側受光，路燈與樹木的陰影向右拉長。
// 動作／視覺規格:
//   - 場景 1：地平線位於 y 150，前方為頭肩人物，頭部中心 x 270、y 78、半徑 29 px，沒有臉部或頭髮。
//   - 太陽半徑 18 px，從 (x 110, y −30) 移至 (x 96, y 136)，一半被山丘遮住。
//   - 色彩：天空頂部 #FAF9F7 → #FFE8BC，地平線附近為 #FFCF80；地面 #F1F0EE → #F6DDB0；人物亮部 #FFB547、陰影 #3E3650。
//   - 頭部左側約 66° 方向受光，邊界右側為陰影；身體左亮右暗。
//   - 陰影從路燈與樹根向右延伸，長度 0 → 150 px，色彩 #DDAA6E；右上時間 12:00 → 17:40。
//   - 時間：0.4 s 到 1.4 s 漸入，維持至 4.0 s，於 4.8 s 前退去；每輪 = 5 s，採 cubic ease-in-out，約為 cubic-bezier(0.65,0,0.35,1)。
// 使用情境: 適合溫暖、懷舊而柔和的影像。
// 關鍵字: golden hour, magic hour, sunset light, warm light, long shadows, low sun
// 尺寸: 539x208 px · 週期: 5 s
// 需要: kit photo -> ./api/v1/kit/zh-Hant/photo.js
// 本地示範: ./?lang=zh-Hant#103
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D，呼叫端先清除畫布並以 save/restore 包覆；w、h 為尺寸，u 為 [0, cycle) 內的秒數。畫面只取決於 u。
// 下方是元素程式碼；CELLS、常數與輔助函式位於 kit。

// @element 07

CELLS[6] = cell((k, u) => {
  const a = env(u), R = PR, W = torsoW(R);
  const GOLD = {
    sky: '#FFE8BC', sky2: '#FFCF80', far: '#F2C58C', mid: '#E8B27A', ground: '#F6DDB0', ground2: '#ECC892', mark: '#F6DDB0', cast: '#DDAA6E',
    bld: '#F7DCAE', edge: '#D9AE76', win: '#ECC892', winOff: '#ECC892', tree: '#E9C08A', trunk: '#C9965E', post: '#B98A58', bulb: '#C9965E',
    person: '#5C5470', shade: '#3E3650', lit: '#FFB547', car: '#A08E8E', glass: '#F6DDB0', tire: '#3E3650', sun: '#F59A24', glow: '#FFD48A'
  };
  const p = pal(BASE, GOLD, a);
  stagePortrait(k, p, { sun: [lerp(110, 96, a), lerp(-30, 136, a), 18, 1], cast: lerp(0, 150, a) });
  const lit = mixc(p.person, GOLD.lit, a), sh = mixc(p.person, GOLD.shade, a);
  bust(k, PX, PY, R, p.person, kk => {
    kk.fillStyle = lit; kk.beginPath(); kk.arc(PX, PY, R, 0, TAU); kk.fill();
    headShade(kk, PX, PY, R, 1.15, 1, sh);
    torso(kk, PX, PY, R, PX - W, PX - 0.5 * W, lit);
    torso(kk, PX, PY, R, PX + 0.4 * W, PX + W, sh);
  });
  hud(k, clock(lerp(720, 1060, a)));
});
