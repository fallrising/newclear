// 長時間曝光（Long exposure）
// 分類組: photo (攝影效果、打光與風格) · element 11 · 分類: 鏡頭與相機效果
// 示範內容: 街景轉為夜晚，移動的汽車幾乎消失，紅色尾燈線在車後延伸，對向車道則出現白色與琥珀色光線。
// 動作／視覺規格:
//   - 場景 2：街角的建築、樹木與路燈（x 114、292、494）位於 y 132–162 的道路後方；人物站在左前方，腳位於 x 176、y 200，身高 124 px。色彩轉為夜景，天空為 #2C3B5E，每三扇窗有一扇為 #FFC75A。
//   - 汽車每 5 s 從 x −60 → 600；車身不透明度 100% → 10%。
//   - 尾燈線從 x −60 延伸至汽車：y 145 為 #FF5B4D、厚 2.6 px；y 149 為 #FF8A6E、厚 1.4 px。
//   - 對向車道的全寬光線：y 135 為 #FFE9B8、厚 2.6 px；y 138.5 為 #FFC75A、厚 1.6 px；y 141.5 為 #FFE9B8、厚 1.2 px，隨效果淡入。
//   - 右上快門速度：1/250 → 8″。
//   - 時間：0.4 s 到 1.4 s 漸入，維持至 4.0 s，於 4.8 s 前退去；每輪 = 5 s，採 cubic ease-in-out，約為 cubic-bezier(0.65,0,0.35,1)。
// 使用情境: 適合在一張靜態照片中呈現車流，或如絲綢般的水與雲。
// 關鍵字: long exposure, light trails, slow shutter, night photography, car trails, motion
// 尺寸: 539x208 px · 週期: 5 s
// 需要: kit photo -> ./api/v1/kit/zh-Hant/photo.js
// 本地示範: ./?lang=zh-Hant#107
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D，呼叫端先清除畫布並以 save/restore 包覆；w、h 為尺寸，u 為 [0, cycle) 內的秒數。畫面只取決於 u。
// 下方是元素程式碼；CELLS、常數與輔助函式位於 kit。

// @element 11

CELLS[10] = cell((k, u) => {
  const a = env(u), x = lerp(-60, 600, u / CYC);
  const p = pal(BASE, NIGHT, a);
  stageStreet(k, p, {
    car: false, person: false,
    after: D => {
      const kk = D.k;

      kk.save(); kk.globalAlpha = a; kk.lineCap = 'round';
      const x0 = -60, x1 = Math.min(x, CW + 40);
      const ln = (ya, yb, c1, c2, w1, w2) => { kk.strokeStyle = c1; kk.lineWidth = w1; kk.beginPath(); kk.moveTo(x0, ya); kk.lineTo(x1, ya); kk.stroke(); kk.strokeStyle = c2; kk.lineWidth = w2; kk.beginPath(); kk.moveTo(x0, yb); kk.lineTo(x1, yb); kk.stroke(); };
      ln(145, 149, '#FF5B4D', '#FF8A6E', 2.6, 1.4);

      for (const [y, c, lw] of [[135, '#FFE9B8', 2.6], [138.5, BASE.lit, 1.6], [141.5, '#FFE9B8', 1.2]]) { kk.strokeStyle = c; kk.lineWidth = lw; kk.beginPath(); kk.moveTo(-10, y); kk.lineTo(CW + 10, y); kk.stroke(); }
      kk.restore();
    }
  });

  k.save(); k.globalAlpha = 1 - 0.9 * a; const D = pen(k);
  car(D, x, p, { head: BASE.lit, tail: '#FF5B4D' }); k.restore();
  figure(pen(k), FIG[0], FIG[1], FIG[2], p.person);
  hud(k, a > 0.5 ? '8″' : '1/250');
});
