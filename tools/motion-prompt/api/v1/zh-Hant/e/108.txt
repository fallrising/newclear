// 追焦攝影（Panning shot）
// 分類組: photo (攝影效果、打光與風格) · element 12 · 分類: 鏡頭與相機效果
// 示範內容: 鏡頭跟隨移動汽車，讓車子保持在畫面中央且清晰，只有後方建築、樹木與路燈橫向拖影。
// 動作／視覺規格:
//   - 街角的建築、樹木與路燈位於 y 132–162 的道路後方，前景沒有大型人物。
//   - 沒有套用效果時，汽車在 5 s 內從 x −80 開到 620；套用後保持在 x 270 ±10 px。
//   - 拖影：將背景沿橫向 46 px 範圍繪製 12 次並取平均，擴散範圍 0 → 46 px。
//   - 輪輻線每 5 s 轉 12 圈。
//   - 右上快門速度：1/1000 → 1/15。
//   - 時間：0.4 s 到 1.4 s 漸入，維持至 4.0 s，於 4.8 s 前退去；每輪 = 5 s，採 cubic ease-in-out，約為 cubic-bezier(0.65,0,0.35,1)。
// 使用情境: 適合在靜態照片中表現汽車、自行車或跑者的速度感。
// 關鍵字: panning, panning shot, motion blur, speed, sports photography, slow shutter
// 尺寸: 539x208 px · 週期: 5 s
// 需要: kit photo -> ./api/v1/kit/zh-Hant/photo.js
// 本地示範: ./?lang=zh-Hant#108
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D，呼叫端先清除畫布並以 save/restore 包覆；w、h 為尺寸，u 為 [0, cycle) 內的秒數。畫面只取決於 u。
// 下方是元素程式碼；CELLS、常數與輔助函式位於 kit。

// @element 12

CELLS[11] = cell((k, u) => {
  const a = env(u);
  const drive = lerp(-80, 620, u / CYC), x = lerp(drive, 270 + 10 * Math.sin(u / CYC * TAU), a);
  const S = 46 * a, n = S < 0.5 ? 1 : 12;
  for (let i = 0; i < n; i++) {
    k.save(); k.globalAlpha = 1 / (i + 1); k.translate(n > 1 ? lerp(-S / 2, S / 2, i / (n - 1)) : 0, 0); stageStreet(k, BASE, { car: false, person: false }); k.restore();
  }
  const D = pen(k); car(D, x, BASE, { head: BASE.lit });

  k.save(); k.strokeStyle = BASE.tire; k.lineWidth = 1.6;
  for (const wx of [x - 21, x + 21]) { const t = u / CYC * TAU * 12; k.beginPath(); k.moveTo(wx - Math.cos(t) * 5, 156 - Math.sin(t) * 5); k.lineTo(wx + Math.cos(t) * 5, 156 + Math.sin(t) * 5); k.stroke(); }
  k.restore();
  hud(k, a > 0.5 ? '1/15' : '1/1000');
});
