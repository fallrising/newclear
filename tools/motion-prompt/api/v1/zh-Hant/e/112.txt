// 使用情境: 適合聚焦主體，或營造老鏡頭的感覺。
// 關鍵字: vignette, vignetting, dark corners, edge darkening, lens falloff, focus
// 尺寸: 539x208 px · 週期: 5 s
// 需要: kit photo -> ./api/v1/kit/zh-Hant/photo.js
// 本地示範: ./?lang=zh-Hant#112
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D，呼叫端先清除畫布並以 save/restore 包覆；w、h 為尺寸，u 為 [0, cycle) 內的秒數。畫面只取決於 u。
// 下方是元素程式碼；CELLS、常數與輔助函式位於 kit。

// @element 16

CELLS[15] = cell((k, u) => {
  const a = env(u);
  stageStreet(k, BASE);

  k.save(); k.translate(CW / 2, CH / 2); k.scale(1, CH / CW * 1.35);
  const g = k.createRadialGradient(0, 0, CW * 0.45 * 0.62, 0, 0, CW * 0.62);
  g.addColorStop(0, 'rgba(32,35,43,0)'); g.addColorStop(1, `rgba(32,35,43,${(0.7 * a).toFixed(3)})`);
  k.fillStyle = g; k.fillRect(-CW, -CW, CW * 2, CW * 2); k.restore();
});
