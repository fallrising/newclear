// 雙重曝光（Double exposure）
// 分類組: photo (攝影效果、打光與風格) · element 24 · 分類: 色調與後製風格
// 示範內容: 背景轉白，山丘、房屋、樹木與太陽的景色縮小後，只出現在胸像輪廓內。
// 動作／視覺規格:
//   - 場景 3：左側為胸像，頭部中心 x 140、y 82、半徑 28 px；右側有山丘、房屋、樹木與太陽，太陽位於 x 432、y 54、半徑 15 px。
//   - 胸像外的背景為白色 #FFFFFF。
//   - 胸像內使用同一片景色的 0.7× 縮小版，讓太陽位於頭部右上方，山稜橫跨頸部。
//   - 內部色彩：天空 #D3DBE6，山丘 #9AA8BD 與 #76869F，地面 #55647E，樹木 #3F4C66。
//   - 轉換時依效果強度，將雙重曝光影像疊在原圖上。
//   - 時間：0.4 s 到 1.4 s 漸入，維持至 4.0 s，於 4.8 s 前退去；每輪 = 5 s，採 cubic ease-in-out，約為 cubic-bezier(0.65,0,0.35,1)。
// 使用情境: 適合海報或封面，將人物與地點、記憶或內心世界連結。
// 關鍵字: double exposure, multiple exposure, blend, silhouette, surreal, composite
// 尺寸: 539x208 px · 週期: 5 s
// 需要: kit photo -> ./api/v1/kit/zh-Hant/photo.js
// 本地示範: ./?lang=zh-Hant#120
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D，呼叫端先清除畫布並以 save/restore 包覆；w、h 為尺寸，u 為 [0, cycle) 內的秒數。畫面只取決於 u。
// 下方是元素程式碼；CELLS、常數與輔助函式位於 kit。

// @element 24

CELLS[23] = cell((k, u) => {
  const a = env(u);
  const DX = Object.assign({}, BASE, { sky: '#D3DBE6', sky2: '#D3DBE6', far: '#9AA8BD', mid: '#76869F', ground: '#55647E', bld: '#E8ECF2', edge: '#55647E', win: '#9AA8BD', winOff: '#9AA8BD', tree: '#3F4C66', trunk: '#3F4C66' });
  const draw = kk => {
    kk.fillStyle = '#FFFFFF'; kk.fillRect(0, 0, CW, CH);
    kk.save(); bustPath(kk, QX, QY, QR); kk.clip();

    kk.translate(QX, QY); kk.scale(0.7, 0.7); kk.translate(-415, -77);
    kk.fillStyle = DX.sky; kk.fillRect(0, -200, CW * 2, CH * 3);
    stageLand(kk, DX, { person: false });
    kk.fillStyle = DX.ground; kk.fillRect(-CW, 200, CW * 3, CH * 3);
    kk.restore();
  };
  if (a >= 0.999) { draw(k); return; }
  stageLand(k, BASE);
  if (a < 0.002) return;
  const [L, kl] = layer('dx'); draw(kl);
  k.save(); k.globalAlpha = a; k.drawImage(L, 0, 0); k.restore();
});
